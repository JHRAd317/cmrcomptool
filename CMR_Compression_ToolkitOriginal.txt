function folderClick {

function fileClick([string]$initalDirectory) {
    #Select File to compress
    #Write-Host "---------------------------------------------------------"
    #Write-Host "File Click Function"

    #Write-Host $initalDirectory
    #Write-Host $folderPath
  
    $null = Add-Type -AssemblyName System.windows.forms
    
    $OpenFileDialog = New-Object -TypeName System.Windows.Forms.OpenFileDialog
    $OpenFileDialog.InitialDirectory = $folderBox.Text
    $OpenFileDialog.filter = 'All files (*.*)| *.*'
    if($OpenFileDialog.ShowDialog() -eq "OK")
    {
        $fileBox.Text = $OpenFileDialog.filename
        Write-Host "Selected File:" $OpenFileDialog.filename
    }else {
        Write-Host "No file selected"
        return
    }

    $file = Get-Item $OpenFileDialog.FileName

    Write-Host "-----------------------"

    Write-Host $file.Name
    Write-Host $file.FullName
    Write-Host $file.Directory
    Write-Host $file.DirectoryName
    $folderToSearch = $file.DirectoryName

    $prog1Label.Visible = $true
    $objShell = New-Object -ComObject Shell.Application
    $objFolder = $objShell.Namespace($folderToSearch)

    Write-Host $objFolder
    
    $filenameWithExtension = $file.Name
    Write-Host $filenameWithExtension

    $progressBar.Text = "Analyzing file..."
    
    

    $objFile = $objFolder.ParseName($filenameWithExtension)
    Write-Host $objFile
    Write-Host "-----------------------"
    $fileMeta = [ordered]@{}
    for($id = 0; $id -le 500; $id++){
        $fileMeta[ $($objFolder.GetDetailsOf($objFolder, $id)) ] = $(
            if($objFolder.GetDetailsOf($objFile, $id)){
                $($objFolder.GetDetailsOf($objFile, $id))
                $progressBar.Value = $id
            }else{ "" }
        )
    }
    
    # print ordered hashtable
    #$fileMeta
    #$fileMeta.Count
    
    $nameText = $OpenFileDialog.FileName
    $drText = "Data Rate (Kbps): $($fileMeta['Data rate'])"
    $fileSizeText = "File Size (MB): $([math]::Round($file.Length / 1MB,2))"
    $fileSizeRaw = $file.Length
    $dataLabelSourceSize.Text = $fileSizeRaw

    $fileNameLabel.Text = $nameText
    $fileDRLabel.Text = $drText
    $fileSizeLabel.Text = $fileSizeText

    $dr = $fileMeta['Data rate']
    $tbr = $fileMeta['Total bitrate']
    $drIndex = $dr.IndexOf('k')

    $drSubString = ($dr.Substring(0,$drIndex))
    $tbrSubString = ($tbr.Substring(0,$drIndex))
    $dataRateIn = [int]::Parse($drSubString.Replace([char] 0x200e,' ').Trim())
    $tbrIn = [int]::Parse($tbrSubString.Replace([char] 0x200e,' ').Trim())
    #$scaleCalc = 400 / $([math]::Round($file.Length / 1MB,2))
    $scaleCalc = 400 / $([math]::Ceiling($file.Length / 1MB))
    #Write-Host $dr
    #Write-Host $drIndex
    #Write-Host $dataRateIn
    #Write-Host $drSubString
    $dataRateCalc = $([math]::Floor(($tbrIn * $scaleCalc)))
    $dataRateCalcString = [string]$dataRateCalc
    $dataRateCalcString = $dataRateCalcString.Replace('.','_')
    $drOutNumLabel.Text = $dataRateCalc

    $dataLabelSourceDR.Text = $dataRateIn
    $dataLabelSourceTBR.Text = $tbrIn
    $dataLabelCalcScale.Text = $scaleCalc
    $dataLabelCalcDR.Text = $dataRateCalc
    $dataLabelSourceWidth.Text = $fileMeta['Frame width']
    $dataLabelSourceHeight.Text = $fileMeta['Frame height']

    $fileABR = $fileMeta['Bit rate']
    $abrSubString = ($fileABR.Substring(0,$fileABR.IndexOf('k')))
    $abrIn = $abrSubString.Replace([char] 0x200e,' ').Trim()
    $dataLabelSourceABR.Text = $abrIn
    
    $fileLength = $fileMeta['Length']
    $fileLengthArray = $fileLength.Split(":")
    $fileLengthSeconds = ( ([int]::Parse($fileLengthArray[0])*60*60) + ([int]::Parse($fileLengthArray[1])*60) + [int]::Parse($fileLengthArray[2]) ) 
    $dataLabelSourceLength.Text = $fileLengthSeconds

    $fileSaveName = $nameText -replace "[.,;/?()`~'""]","_"
    #$fileSaveName = $fileSaveName.Replace(' ','_')
    #$fileSaveName = $fileSaveName.Replace('.','_')
    #Write-Host $fileSaveName
    
    $fileSaveName = "$fileSaveName-DR-$dataRateCalcString.mp4"
    $saveBox.Text = "$fileSaveName"
    $calcLabel.Text = "Calculation rate: $scaleCalc"
    $drOutLabel.Text = "Output Data Rate: $dataRateCalc"

    $prog2Label.Visible = $false
    $prog3Label.Visible = $false

    $progressBar.Text = "File Ready To Compress"

    return $OpenFileDialog.FileName



}

Function sendToVLC {
    #Send compression file to VLC
    $vlcButton.Enabled = $false
    $outProgressBar.Value = 0

    $prog2Label.Visible = $true
    $vlcPath = "C:\Program Files\VideoLAN\VLC\vlc.exe"

    $fileToCompressName = $fileNameLabel.Text
    $destinationPath = $saveBox.Text
    
    
    $dataRateCalc = $([math]::Floor($drOutNumLabel.Text))

    $vlcButton.Visible = $false

    $dataLabelAttemptDateTime.Text = Get-Date -Format "yyyy-MM-dd_HH:mm:ss"
    $dataLabelStartTime.Text = Get-Date -Format "HH:mm"
    #$dataLabelCalcDR.Text = $dataRateCalcs

    Write-Host "$fileToCompressName | Start Time:  $(Get-Date -Format "HH:mm K")| Output: $destinationPath"
    Start-Process $vlcPath -ArgumentList """$fileToCompressName""", "--sout=""#transcode{vcodec=h264,vb=$dataRateCalc}:standard{access=file,mux=mp4,dst=$destinationPath }"", vlc://quit " -wait
    $outFile = Get-Item $destinationPath
    Write-Host "  End Time: $(Get-Date -Format "HH:mm K") Output Size: (MB): $([math]::Round($outFile.Length / 1MB,2))"
    $dataLabelEndTime.Text = Get-Date -Format "HH:mm"


    #AnalyzeOutputFile
    $outObjShell = New-Object -ComObject Shell.Application
    $outObjFolder = $outObjShell.Namespace($outFile.DirectoryName)
    $outObjFile = $outObjFolder.ParseName($outFile.Name)
    $outFileMeta = [ordered]@{}
    for($id = 0; $id -le 500; $id++){
        $outFileMeta[ $($outObjFolder.GetDetailsOf($outObjFolder, $id)) ] = $(
            if($outObjFolder.GetDetailsOf($outObjFile, $id)){
                $($outObjFolder.GetDetailsOf($outObjFile, $id))
                $outProgressBar.Value = $id
            }else{ "" }
        )
    }

    $dataLabelOutSize.Text = $outFile.Length

    

    $outFileABR = $outFileMeta['Bit rate']
    $abrSubString = ($outFileABR.Substring(0,$outFileABR.IndexOf('k')))
    $abrOut = $abrSubString.Replace([char] 0x200e,' ').Trim()
    $dataLabelOutABR.Text = $abrOut

    
    $fileLength = $outFileMeta['Length']
    $fileLengthArray = $fileLength.Split(":")
    $fileLengthSeconds = ( ([int]::Parse($fileLengthArray[0])*60*60) + ([int]::Parse($fileLengthArray[1])*60) + [int]::Parse($fileLengthArray[2]))
    $dataLabelOutLength.Text = $fileLengthSeconds

    $dataLabelOutWidth.Text = $outFileMeta['Frame width']
    $dataLabelOutHeight.Text = $outFileMeta['Frame height']

    $dr = $outFileMeta['Data rate']
    $tbr = $outFileMeta['Total bitrate']
    $drIndex = $dr.IndexOf('k')

    $drSubString = ($dr.Substring(0,$drIndex))
    $tbrSubString = ($tbr.Substring(0,$drIndex))
    $dataRateOut = [int]::Parse($drSubString.Replace([char] 0x200e,' ').Trim())
    $tbrOut = [int]::Parse($tbrSubString.Replace([char] 0x200e,' ').Trim())

    $dataLabelOutDR.Text = $dataRateOut
    $dataLabelOutTBR.Text = $tbrOut

    logAttempt

    $vlcButton.Visible = $true

    $outForm = New-Object System.Windows.Forms.Form
    $outForm.Text = 'Compression Complete'
    $outForm.Size = New-Object System.Drawing.Size(300,50)
    $outForm.StartPosition = 'CenterScreen'
    $outForm.Topmost = $true
    $outCloseButton = New-Object System.Windows.Forms.Button
    $outCloseButton.Location = New-Object System.Drawing.Point(100,25)
    $outCloseButton.Size = New-Object System.Drawing.Size(100,23)
    $outCloseButton.Text = 'Close'
    $outCloseButton.DialogResult = [System.Windows.Forms.DialogResult]::Yes
    $outForm.AcceptButton = $outCloseButton
    $outForm.Controls.Add($outCloseButton)

    $prog3Label.Visible = $true
    $vlcButton.Enabled = $true
}



Function analyzeDrive() {
    #Analyze Drive for files
    $folder = folderClick
    $analyzeCountLabel.Text = "Analyzing..."
    Get-ChildItem -Path $folder -Recurse -File -ErrorAction SilentlyContinue | Where-Object {$_.Length -gt (500 * 1MB) -and  $_.Extension -in @(".mp4", ".wmv", ".avi")} | Sort-Object -Property Length -Descending | Select-Object -OutVariable returnData -Property Name, @{Name='Size (MB)';Expression={[Math]::Round($_.Length/1MB,3)}}, FullName

    $returnData.Count

     
    #Write-Host $tabPageAnalyze.BackColor
    #Write-Host $analyzeFolderButton.BackColor

    if($returnData.Count -gt 0)
    {
        $tabPageAnalyze.BackColor = [System.Drawing.Color]::Red
        $analyzeCountLabel.Text = "$($returnData.Count) Files Require Compression"

    }else {
        $tabPageAnalyze.BackColor = [System.Drawing.Color]::Green
        $analyzeCountLabel.Text = "No Files Require Compression"
    }


}
